📋 SETUP (5 MINUTES)

1. TELEGRAM
   ➜ Chat with @BotFather on Telegram
   ➜ /newbot → Choose name/username → Copy TOKEN
   ➜ Visit: https://api.telegram.org/bot<TOKEN>/getUpdates
   ➜ Find "chat":{"id":<CHAT_ID>}
   ➜ Add to .env:
     TELEGRAM_BOT_TOKEN=<token>
     TELEGRAM_CHAT_ID=<id>

2. COINGECKO
   ➜ No setup needed! Works instantly
   ➜ Optional: Add API key for more calls
     COINGECKO_API_KEY=<key>

3. BITGET
   ➜ Create account: https://www.bitget.com/
   ➜ Settings → API → Create key
   ➜ Add to .env:
     BITGET_API_KEY=<key>
     BITGET_SECRET=<secret>
     BITGET_API_PASSPHRASE=<pass>

4. INSTALL
   pip install -r requirements.txt -U

5. TEST
   python test_integrations.py

🚀 QUICK USAGE

# Get market data from all sources
python integrated_data_pipeline.py

# Send Telegram alert
python -c "
import asyncio
from data_sources.telegram_notifier import TelegramNotifier

async def test():
    t = TelegramNotifier()
    await t.send_alert('Test', 'Hello from AiTradingAgent!', 'SUCCESS')

asyncio.run(test())
"

# Get CoinGecko data
python -c "
from data_sources.coingecko_feed import CoinGeckoFeed
cg = CoinGeckoFeed()
print(f'BTC: {cg.get_price(\"bitcoin\")}')
print(f'Trending: {[t[\"item\"][\"name\"] for t in cg.get_trending(3)]}')
"

# Get Bitget balance
python -c "
from data_sources.bitget_exchange import BitgetExchange
bg = BitgetExchange(sandbox=True)
print(f'Connected: {bg.is_connected()}')
print(f'Balance: {bg.get_total_balance_usd()}')
"
